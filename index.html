function renderProducts() {

  const cakes = products.filter(p => p.type === "cake");

  // 蛋糕

  let cakeHTML = "";

  cakes.forEach(p => {

      sizesHTML += `

        <div class="size-row" id="row-${key}">

              <button class="qty-btn" onclick="changeCake('${key}',1,${s.price})">+</button>

            </div>

          </div>

        </div>`;

    });


    cakeHTML += `

      <div class="product-card">

        ${imgHTML}

        <div class="product-info">

// 數量控制

// ==============================

function changeCake(key, delta) {

  cakeQty[key] = Math.max(0, (cakeQty[key] || 0) + delta);

  document.getElementById(`qty-${key}`).textContent = cakeQty[key];

  const row = document.getElementById(`row-${key}`);

  if (row) row.className = "size-row" + (cakeQty[key] > 0 ? " has-item" : "");

  updateSummary();

}


function changeSingle(id, delta) {

  singleQty[id] = Math.max(0, (singleQty[id] || 0) + delta);

  document.getElementById(`qty-single-${id}`).textContent = singleQty[id];

  const row = document.getElementById(`single-row-${id}`);

  if (row) row.className = "addon-row" + (singleQty[id] > 0 ? " has-item" : "");

  updateSummary();

}


function changeAddon(key, delta) {

  addonQty[key] = Math.max(0, addonQty[key] + delta);

  document.getElementById(`qty-${key}`).textContent = addonQty[key];

  const row = document.getElementById(`${key}-row`);

  if (row) row.className = "addon-row" + (addonQty[key] > 0 ? " has-item" : "");

  updateSummary();

}


// ==============================

// 金額摘要

// ==============================

function updateSummary() {

  let total = 0, html = "";


  // 蛋糕

  products.filter(p => p.type === "cake").forEach(p => {

    p.sizes.forEach(s => {

      const key = `${p.id}_${s.size}`;

      const q = cakeQty[key] || 0;

      if (q > 0) {

        const sub = s.price * q;

        html += `<div class="s-row"><span>${p.name} ${s.size}${q > 1 ? ` ×${q}` : ""}</span><span>NT$${sub.toLocaleString()}</span></div>`;

        total += sub;

      }

    });

  });


  // 單品

  products.filter(p => p.type === "single").forEach(p => {

    const q = singleQty[p.id] || 0;

    if (q > 0) {

      const sub = p.price * q;

      html += `<div class="s-row"><span>${p.name} ×${q}</span><span>NT$${sub.toLocaleString()}</span></div>`;

      total += sub;

    }

  });


  // 餐具

  const plateQ = parseInt(document.getElementById("plate").value || 0);

  if (plateQ > 0) {

    const sub = plateQ * 30;

    html += `<div class="s-row"><span>餐具 ×${plateQ}組</span><span>NT$${sub.toLocaleString()}</span></div>`;

    total += sub;

  }


  // 保冷袋

  if (addonQty.bag > 0) {

    const sub = addonQty.bag * 100;

    html += `<div class="s-row"><span>保冷袋 ×${addonQty.bag}</span><span>NT$${sub.toLocaleString()}</span></div>`;

    total += sub;

  }


  document.getElementById("summaryItems").innerHTML = html || '<div style="color:#6a5040;font-size:12px;text-align:center">尚未選擇任何品項</div>';

  document.getElementById("summaryTotal").textContent = `NT$${total.toLocaleString()}`;

}

// 驗證

// ==============================

function validate() {

  let ok = true;

  [["name","err-name"],["phone","err-phone"],["pickupDate","err-date"],["pickupTime","err-time"]].forEach(([id,err]) => {

    const v = document.getElementById(id).value.trim();

    document.getElementById(err).style.display = v ? "none" : "block";

    if (!v) ok = false;

  });


  // 至少選一項

  const hasCake = Object.values(cakeQty).some(q => q > 0);

  const hasSingle = Object.values(singleQty).some(q => q > 0);

  if (!hasCake && !hasSingle) { alert("請至少選擇一項商品！"); ok = false; }

  return ok;

}


// ==============================

// 組合訂單資料

// ==============================

function buildOrderData() {

  let total = 0;

  const items = [];


  products.filter(p => p.type === "cake").forEach(p => {

    p.sizes.forEach(s => {

      const q = cakeQty[`${p.id}_${s.size}`] || 0;

      if (q > 0) {

        items.push({ name:`${p.name} ${s.size}`, qty: q, price: s.price, sub: s.price * q });

        total += s.price * q;

      }

    });

  });


  products.filter(p => p.type === "single").forEach(p => {

    const q = singleQty[p.id] || 0;

    if (q > 0) {

      items.push({ name: p.name, qty: q, price: p.price, sub: p.price * q });

      total += p.price * q;

    }

  });


  const plateQ = parseInt(document.getElementById("plate").value || 0);

  if (plateQ > 0) { items.push({ name:`餐具`, qty: plateQ, price: 30, sub: plateQ * 30 }); total += plateQ * 30; }

  if (addonQty.bag > 0) { items.push({ name:"保冷袋", qty: addonQty.bag, price: 100, sub: addonQty.bag * 100 }); total += addonQty.bag * 100; }


  return {

    lineUID,

    name: document.getElementById("name").value.trim(),

    phone: document.getElementById("phone").value.trim(),

    pickupDate: document.getElementById("pickupDate").value,

    pickupTime: document.getElementById("pickupTime").value,

    items,

    total,

    note: document.getElementById("note").value.trim(),

    timestamp: new Date().toLocaleString("zh-TW", { timeZone: "Asia/Taipei" })

  };

}

async function submitForm() {

  if (!validate()) return;

  const btn = document.getElementById("submitBtn");

  btn.disabled = true;

  btn.textContent = "送出中⋯⋯";


  try {

    await fetch(GAS_URL, {

      method: "POST",

      mode: "no-cors",

      headers: { "Content-Type": "application/json" },
